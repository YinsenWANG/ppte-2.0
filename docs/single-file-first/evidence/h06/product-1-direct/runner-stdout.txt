(function(){
  "use strict";
  const pages=Array.from(document.querySelectorAll(".slide"));
  const counter=document.getElementById("counter");
  const present=document.getElementById("present");
  let current=0,timer;
  function update(i){
    current=i;
    counter.textContent=String(i+1).padStart(2,"0")+" / "+pages.length;
  }
  function go(i){
    i=Math.max(0,Math.min(pages.length-1,i));
    update(i);
    pages[i].scrollIntoView({behavior:matchMedia("(prefers-reduced-motion: reduce)").matches?"instant":"smooth",block:"start"});
  }
  function reveal(){
    document.body.classList.add("controls");
    clearTimeout(timer);
    timer=setTimeout(function(){document.body.classList.remove("controls")},1800);
  }
  function setPresentation(on){
    document.body.classList.toggle("present",on);
    document.body.classList.remove("blackout");
    present.setAttribute("aria-pressed",String(on));
    present.textContent=on?"退出放映":"放映本稿";
    if(on){present.blur();go(current);reveal();}
  }
  document.getElementById("prev").addEventListener("click",function(){go(current-1)});
  document.getElementById("next").addEventListener("click",function(){go(current+1)});
  present.addEventListener("click",function(){setPresentation(!document.body.classList.contains("present"))});
  const observer=new IntersectionObserver(function(entries){
    let best=null;
    entries.forEach(function(entry){if(entry.isIntersecting&&(!best||entry.intersectionRatio>best.intersectionRatio))best=entry});
    if(best)update(pages.indexOf(best.target));
  },{threshold:[.5,.65,.85]});
  pages.forEach(function(p){observer.observe(p)});
  document.addEventListener("pointermove",function(){if(document.body.classList.contains("present"))reveal()});
  document.addEventListener("keydown",function(event){
    if(event.altKey||event.ctrlKey||event.metaKey)return;
    if(event.key==="Escape"){setPresentation(false);return}
    if((event.key==="b"||event.key==="B")&&document.body.classList.contains("present")){
      event.preventDefault();document.body.classList.toggle("blackout");return;
    }
    if(document.body.classList.contains("blackout"))return;
    if(event.target.closest("button,input,textarea,select,[contenteditable]"))return;
    if(["ArrowRight","ArrowDown","PageDown"," "].includes(event.key)){event.preventDefault();go(current+1)}
    if(["ArrowLeft","ArrowUp","PageUp"].includes(event.key)){event.preventDefault();go(current-1)}
    if(event.key==="Home"){event.preventDefault();go(0)}
    if(event.key==="End"){event.preventDefault();go(pages.length-1)}
  });
})();

<div class="status-track" aria-label="设计目标状态流程">
<span class="status">已保存</span><span class="arrow">→</span><span class="status">有修改</span><span class="arrow">→</span><span class="status">保存中</span><span class="arrow">→</span><span class="status current">落盘确认 · 已保存</span>

<div class="two-cols">
<div class="panel"><div class="eyebrow">作者看得懂</div><h3 style="margin-top:1cqw">草稿，不冒充文件。</h3><p style="margin-top:1cqw;font-size:1.3cqw">无文件授权、仅草稿、冲突与失败分别表达。失败时保留修改，状态持续可见。</p><p class="small muted" style="margin-top:1cqw">主操作为“保存”；Cmd / Ctrl + S 可重试。</p></div>
<div class="panel"><div class="eyebrow">写入有秩序</div><h3 style="margin-top:1cqw">发现冲突，停止覆盖。</h3><p style="margin-top:1cqw;font-size:1.3cqw">串行写入并绑定版本；保存前检查磁盘变化。外部修改出现时，提供重新读取或保留当前修改。</p><p class="small muted" style="margin-top:1cqw">中文输入组合期间，不提交半个汉字。</p></div>

<p class="small muted" style="margin-top:1.5cqw">可靠性目标：本地服务校验后替换文件，保留可恢复版本。浏览器直写仍存在与外部进程并发的边界；断电耐久性必须通过目标系统实测确定。</p>

(function(){

let index=0,playing=false,timer;
function update(){
slides.forEach((slide,i)=>slide.classList.toggle("shown",i===index));
counter.textContent=String(index+1).padStart(2,"0")+" / "+slides.length;
prev.disabled=index===0;
next.disabled=index===slides.length-1;
}
function go(n){
index=Math.max(0,Math.min(slides.length-1,n));
update();
if(!playing)slides[index].scrollIntoView({behavior:"auto",block:"center"});
}
function reveal(){
document.body.classList.remove("quiet");
clearTimeout(timer);
if(playing)timer=setTimeout(()=>document.body.classList.add("quiet"),2200);
}
function toggle(){
playing=!playing;
document.body.classList.toggle("present",playing);
document.body.classList.remove("black","quiet");
present.textContent=playing?"退出放映":"放映";
present.setAttribute("aria-pressed",String(playing));
update();
if(!playing){
clearTimeout(timer);
slides[index].scrollIntoView({behavior:"auto",block:"center"});
}else reveal();
}
prev.addEventListener("click",()=>go(index-1));
next.addEventListener("click",()=>go(index+1));
present.addEventListener("click",toggle);
document.addEventListener("mousemove",reveal);
document.addEventListener("pointerdown",()=>{if(playing)reveal();});
document.addEventListener("focusin",reveal);
document.addEventListener("keydown",e=>{
if(e.key==="Escape"){
document.body.classList.remove("black");
if(playing)toggle();
return;
}
if(!playing)return;
if(e.key.toLowerCase()==="b"){
e.preventDefault();
document.body.classList.toggle("black");
return;
}
if(document.body.classList.contains("black"))return;
if(e.target.closest("button")&&(e.key===" "||e.key==="Enter"))return;
if(["ArrowRight","ArrowDown","PageDown"," "].includes(e.key)){e.preventDefault();go(index+1);}
if(["ArrowLeft","ArrowUp","PageUp"].includes(e.key)){e.preventDefault();go(index-1);}
if(e.key==="Home"){e.preventDefault();go(0);}
if(e.key==="End"){e.preventDefault();go(slides.length-1);}
if(e.key==="Tab")reveal();
});
const observer=new IntersectionObserver(entries=>{
if(playing)return;
const visible=entries.filter(e=>e.isIntersecting).sort((a,b)=>b.intersectionRatio-a.intersectionRatio);
if(visible.length){index=slides.indexOf(visible[0].target);update();}
},{threshold:[.35,.6,.85]});
